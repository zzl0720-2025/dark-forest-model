 167 | def replay_payload(config, results):
 168 |     kind_code={"birth":0,"collapse":1,"attack":2,"empty_search":3,"undetected_search":4}
 169 |     runs=[]
 170 |     for p in config["replay_probabilities"]:
 171 |         trial=results[p][0]  # The first repetition, selected before seeing outcomes.
 172 |         runs.append({"p":p,"seed":trial.seed,"initial":trial.initial_sites,
 173 |                      "events":[[round(e.time,7),kind_code[e.kind],e.source,e.target,e.count_after] for e in trial.events],
 174 |                      "ode":[round(trial.parameters.capacity*ode_solution(t,trial.parameters.initial/trial.parameters.capacity,trial.parameters),6)
 175 |                             for t in trial.sample_times],
 176 |                      "times":trial.sample_times,"counts":trial.sample_counts})
 177 |     return {"side":config["side"],"duration":config["duration"],"runs":runs}

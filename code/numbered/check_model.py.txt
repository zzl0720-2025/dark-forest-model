  64 |     def test_stationary_detailed_balance(self):

  68 |             self.assertAlmostEqual(sum(law), 1)
  69 |             for n in range(p.capacity):
  70 |                 up = law[n]*p.birth*(p.capacity-n)
  71 |                 down = law[n+1]*(p.collapse*(n+1)+p.search*prob*(n+1)*n/(p.capacity-1))

  73 |             if prob == 0:
  74 |                 q = p.birth/(p.birth+p.collapse)
